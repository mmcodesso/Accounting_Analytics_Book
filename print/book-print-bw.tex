% The black-and-white print interior: every color a gray (IngramSpark: grayscale, text in 100% black).
% The values are LATEX_COLORS in scripts/print/palette.py, which also gives the figures their gray levels;
% tests/test_print.py checks that the two agree. Tints are at least 12% black, because the screen's
% 5-6% tints can vanish on the press.

% Colors used by name after this point (Quarto's callout colors, lua-ul's shading) are converted to gray.
\selectcolormodel{gray}

% Colors defined before this point keep their RGB values when tcolorbox or pgf copy them, so each is
% defined again as a gray. Headings, captions and rules are black.
\definecolor{Ink}{gray}{0}
\definecolor{PrimaryBlue}{gray}{0}
\definecolor{Teal}{gray}{0.3}
\definecolor{Coral}{gray}{0.3}
\definecolor{Amber}{gray}{0.5}
\definecolor{LightGray}{gray}{0.88}
\definecolor{BorderGray}{gray}{0.6}
\definecolor{shadecolor}{gray}{0.88}
\definecolor{ObjectivesTint}{gray}{0.88}

% Body text in 100% black.
\color{Ink}

% Callouts: a black left rule and outline, with a title band at 10% black (Quarto draws it as a 10% tint
% of the callout color). Quarto defines these colors after this header, hence \AtBeginDocument.
\AtBeginDocument{%
  \definecolor{quarto-callout-note-color}{gray}{0}%
  \definecolor{quarto-callout-tip-color}{gray}{0}%
  \definecolor{quarto-callout-warning-color}{gray}{0}%
  \definecolor{quarto-callout-important-color}{gray}{0}%
  \definecolor{quarto-callout-caution-color}{gray}{0}%
  \definecolor{quarto-callout-note-color-frame}{gray}{0.4}%
  \definecolor{quarto-callout-tip-color-frame}{gray}{0.4}%
  \definecolor{quarto-callout-warning-color-frame}{gray}{0.4}%
  \definecolor{quarto-callout-important-color-frame}{gray}{0.4}%
  \definecolor{quarto-callout-caution-color-frame}{gray}{0.4}%
}
